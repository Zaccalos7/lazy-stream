cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" ORBIS_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
if(NOT ORBIS_VERSION MATCHES "^[0-9]+\.[0-9]+\.[0-9]+$")
  message(FATAL_ERROR "linux/VERSION deve contenere MAJOR.MINOR.PATCH")
endif()
project(OrbisStreamLinux VERSION ${ORBIS_VERSION} LANGUAGES CXX)
configure_file(src/version.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/version.hpp" @ONLY)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Questo progetto richiede Linux")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(Boost 1.74 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
find_package(SQLite3 REQUIRED)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets Concurrent PrintSupport)
set(CMAKE_AUTOMOC ON)
add_library(orbis-core src/core.cpp src/database_schema.cpp src/scene_service.cpp src/history_cleanup_service.cpp src/live_rows.cpp src/stream.cpp src/stream_platform.cpp src/live_stream_keys.cpp src/metrics.cpp)
target_include_directories(orbis-core PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated" ${Boost_INCLUDE_DIRS})
target_link_libraries(orbis-core PUBLIC SQLite::SQLite3 PkgConfig::JSONC Threads::Threads)
target_compile_options(orbis-core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(orbis-core PUBLIC BOOST_ERROR_CODE_HEADER_ONLY)
add_executable(orbis-stream-runner src/process_runner.cpp)
add_dependencies(orbis-core orbis-stream-runner)
add_library(orbis-native src/native_i18n.cpp src/native_window.cpp src/native_dialogs.cpp src/native_composer.cpp src/native_media.cpp src/media_job.cpp)
target_compile_options(orbis-native PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(orbis-native PUBLIC orbis-core Qt6::Widgets Qt6::MultimediaWidgets Qt6::Multimedia Qt6::Concurrent Qt6::PrintSupport)
add_executable(orbis-stream src/main.cpp src/server.cpp)
target_link_libraries(orbis-stream PRIVATE orbis-native)
target_compile_definitions(orbis-stream PRIVATE ORBIS_WEB_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/web")
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(orbis-tests tests/core_tests.cpp)
  target_link_libraries(orbis-tests PRIVATE orbis-core)
  add_test(NAME core COMMAND orbis-tests)
  add_executable(orbis-native-tests tests/native_tests.cpp)
  target_link_libraries(orbis-native-tests PRIVATE orbis-native Qt6::Test)
  target_compile_definitions(orbis-native-tests PRIVATE ORBIS_TEST_RESOURCES="${CMAKE_CURRENT_SOURCE_DIR}/web/resources" ORBIS_TEST_SCREENSHOT="${CMAKE_CURRENT_BINARY_DIR}/native-window.png")
  add_test(NAME native-gui COMMAND orbis-native-tests)
  set_tests_properties(native-gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME native-launch COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/native_launch.py $<TARGET_FILE:orbis-stream>)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:orbis-stream>)
  add_test(NAME porting-map COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/porting_map.py --check)
endif()
include(GNUInstallDirs)
install(TARGETS orbis-stream orbis-stream-runner RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY web/ DESTINATION ${CMAKE_INSTALL_DATADIR}/orbis-stream/web)
install(FILES packaging/orbis-stream.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
set(CPACK_GENERATOR "TGZ;DEB")
set(CPACK_PACKAGE_CONTACT "OrbisStream")
set(CPACK_DEBIAN_PACKAGE_NAME "orbis-stream")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "ffmpeg, gstreamer1.0-libav, gstreamer1.0-plugins-good, libsqlite3-0, libjson-c5, curl, libqt6widgets6 (>= ${Qt6_VERSION}) | libqt6widgets6t64 (>= ${Qt6_VERSION}), libqt6multimedia6 (>= ${Qt6_VERSION}), libqt6multimediawidgets6 (>= ${Qt6_VERSION}), libqt6printsupport6 (>= ${Qt6_VERSION}) | libqt6printsupport6t64 (>= ${Qt6_VERSION}), qt6-qpa-plugins, qt6-translations-l10n, fonts-noto-core, fonts-noto-cjk")
set(CPACK_DEBIAN_PACKAGE_SECTION "video")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "OrbisStream: local video streaming studio for Linux")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
include(CPack)
